# 循环，每次找到长度为k的链表，执行翻转. 翻转只需要头节点就能翻转（这里翻转别设置尾巴是None了），翻转完pt跳过k个节点（可以直接pt指向pt.next，因为翻转完原pt.next就是K链表的尾巴，下次继续回到循环K组的起点）
# 依旧把head先接到新节点后面，方便处理head节点

def solve(head,k):
  def rev(begin,end):
    # 1,2,3,4
    if not begin:return begin
    next = rev(begin.next,end) # 4,3,2
    begin.next.next = head
    head.next = end.next
    return next    

  if not head: return head
  
  newh = ListNode(0)
  newh.next = head
  pt = newh
  while pt:
    # 每轮循环找到长度k一组
    cnt = 0
    subpt = pt
    while cnt <k and subpt:
      subpt = subpt.next
      cnt += 1
    if not subpt: break # 不够k个，不必拼接，结束
    # 子链表 pt.next ~ subpt
    
    pt.next = rev(pt.next,subpt)
    
    
    
    pt = pt.next
  return pt.next
